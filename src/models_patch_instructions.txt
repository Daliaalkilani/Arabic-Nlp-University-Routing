تعديل على models.py — داخل class Employee، أضيفي السطر الجديد بعد verification_file:

class Employee(db.Model):
    """جدول بيانات الموظفين الإدارية (مرتبط بجدول المستخدمين الأساسي)"""
    id = db.Column(db.Integer, primary_key=True)
    user_id = db.Column(db.Integer, db.ForeignKey('user.id'), nullable=False)
    full_name = db.Column(db.String(100), nullable=False)
    office_dept = db.Column(db.String(100), nullable=False)
    verification_file = db.Column(db.String(200))

    # 🌟 الإضافة الجديدة: هل وافقت الإدارة على هذا الحساب؟ (False = بانتظار المراجعة)
    is_approved = db.Column(db.Boolean, default=False, nullable=False)

    user = db.relationship('User', backref=db.backref('employee_profile', uselist=False))

وإضافة عمود جديد على جدول موجود مسبقاً ما بينعمل تلقائياً بـ db.create_all()،
لازم تشغلي سكربت الترقية المرفق (upgrade_db_add_approval.py) مرة واحدة بعد هذا التعديل.
